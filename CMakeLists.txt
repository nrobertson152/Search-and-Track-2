cmake_minimum_required(VERSION 3.16)
project(search_and_track_2 VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT UNIX)
  message(FATAL_ERROR "Current renderer requires X11 on Linux. Native Windows backend is a future platform target.")
endif()
find_package(X11 REQUIRED)
add_executable(search-and-track-2 src/main.c src/sim.c)
target_link_libraries(search-and-track-2 PRIVATE ${X11_LIBRARIES} m)
target_include_directories(search-and-track-2 PRIVATE ${X11_INCLUDE_DIR})
include(CTest)
if(BUILD_TESTING)
 add_executable(sat-tests tests/test_sim.c src/sim.c)
 target_link_libraries(sat-tests PRIVATE m)
 add_test(NAME sim_logic COMMAND sat-tests)
 add_test(NAME headless_full_shift COMMAND search-and-track-2 --smoke)
endif()
install(TARGETS search-and-track-2 RUNTIME DESTINATION bin)
